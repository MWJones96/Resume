%-------------------------
% Custom commands
%
% Vertical rhythm is set here, in the list parameters, and nowhere else --
% no \vspace at the call sites. Four gaps matter, and each has exactly one
% source, so any of them can be retuned without knock-on effects:
%
%   section rule -> content   \titlespacing's third argument
%   entry -> entry            outer itemsep + the inner list's closing topsep
%   heading -> its bullets    inner topsep
%   bullet -> bullet          inner itemsep

\newcommand{\resumeItem}[2]{%
  \item\small{\textbf{#1}{: #2}}%
}

\newcommand{\resumeSubheading}[4]{%
  \item
    \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      {\small #3} & {\small #4}
    \end{tabular*}%
}

\newcommand{\resumeSubheadingSmall}[2]{%
  \item
    \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2
    \end{tabular*}%
}

\newcommand{\resumeSubheadingBig}[6]{%
  \item
    \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      {\small #3} & {\small #4} \\
      {\small #5} & {\small #6}
    \end{tabular*}%
}

\newcommand{\resumeSubItem}[2]{\resumeItem{#1}{#2}}

\renewcommand{\labelitemi}{}

% One item per role, degree or project. Flush left, so the entry headings
% line up with the section rules above them.
